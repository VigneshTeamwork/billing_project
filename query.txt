-- ============================================================
-- Supplier Master
-- Date: 2026-10-23
-- ============================================================
-- New table, same structure as `customer_list` (sup_ instead of cus_).
-- Safe to run more than once (IF NOT EXISTS).
-- ============================================================

CREATE TABLE IF NOT EXISTS `supplier_list` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `sup_name` varchar(200) DEFAULT NULL,
  `sup_address` text DEFAULT NULL,
  `sup_mob_no` varchar(200) DEFAULT NULL,
  `sup_email_id` varchar(200) DEFAULT NULL,
  `sup_gst_no` varchar(200) DEFAULT NULL,
  `status` tinyint(1) DEFAULT 1,
  `delete_flag` tinyint(1) DEFAULT 0,
  `date_created` datetime NOT NULL DEFAULT current_timestamp(),
  `date_updated` datetime NOT NULL DEFAULT current_timestamp(),
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;


-- ============================================================
-- Purchase Master
-- Date: 2026-10-23
-- ============================================================
-- 3 new tables (same pattern as transaction_list / transaction_products / payment)
-- + 6 settings rows for Purchase GST / Non-GST code numbering.
-- Run AFTER the Supplier Master section. Safe to run more than once.
-- ============================================================

CREATE TABLE IF NOT EXISTS `purchase_list` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` int(11) NOT NULL,
  `code` varchar(100) NOT NULL,
  `invoice_type` varchar(200) DEFAULT NULL,
  `supplier_id` int(11) DEFAULT NULL,
  `supplier_bill_no` varchar(100) DEFAULT NULL,
  `bill_date` date DEFAULT NULL,
  `sub_amount` float(15,2) DEFAULT 0.00,
  `dis_amt` float(15,2) DEFAULT 0.00,
  `cgst` float(15,2) DEFAULT 0.00,
  `sgst` float(15,2) DEFAULT 0.00,
  `cgst_amt` float(15,2) DEFAULT 0.00,
  `sgst_amt` float(15,2) DEFAULT 0.00,
  `amount_with_gst` float(15,2) DEFAULT 0.00,
  `amount` float(15,2) NOT NULL DEFAULT 0.00,
  `status` tinyint(4) NOT NULL DEFAULT 0 COMMENT '0=Pending,1=Received,2=Paid,3=Cancelled',
  `date_created` datetime NOT NULL DEFAULT current_timestamp(),
  `date_updated` datetime NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`),
  KEY `user_id` (`user_id`),
  KEY `supplier_id` (`supplier_id`),
  CONSTRAINT `user_id_fk_pl` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `purchase_products` (
  `purchase_id` int(11) NOT NULL,
  `product_id` int(11) NOT NULL,
  `qty` int(11) NOT NULL DEFAULT 0,
  `price` float(15,2) NOT NULL DEFAULT 0.00,
  `product_dis_type` varchar(200) DEFAULT NULL,
  `product_discount` varchar(200) DEFAULT NULL,
  `product_discount_amt` float(15,2) DEFAULT 0.00,
  `product_total` float(15,2) DEFAULT 0.00,
  KEY `purchase_id` (`purchase_id`),
  KEY `product_id` (`product_id`),
  CONSTRAINT `product_id_fk_pp` FOREIGN KEY (`product_id`) REFERENCES `product_list` (`id`) ON DELETE CASCADE,
  CONSTRAINT `purchase_id_fk_pp` FOREIGN KEY (`purchase_id`) REFERENCES `purchase_list` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE IF NOT EXISTS `purchase_payment` (
  `payment_id` int(11) NOT NULL AUTO_INCREMENT,
  `purchase_id` int(11) NOT NULL,
  `payment_date` date NOT NULL,
  `amount` float(15,2) NOT NULL DEFAULT 0.00,
  `payment_type` varchar(200) NOT NULL,
  `payment_desc` longtext NOT NULL,
  `created_at` datetime NOT NULL DEFAULT current_timestamp(),
  `status` char(15) NOT NULL DEFAULT '0',
  PRIMARY KEY (`payment_id`),
  KEY `purchase_id` (`purchase_id`),
  CONSTRAINT `purchase_id_fk_ppay` FOREIGN KEY (`purchase_id`) REFERENCES `purchase_list` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- ============================================================
-- Invoice Date (user-entered date on invoice create / edit)
-- Date: 2026-10-23
-- ============================================================
-- New column on `transaction_list`. Existing invoices get the date
-- they were created, so their printed date does not change.
-- Safe to run more than once.
-- ============================================================

ALTER TABLE `transaction_list` ADD COLUMN IF NOT EXISTS `invoice_date` date DEFAULT NULL AFTER `code`;
UPDATE `transaction_list` SET `invoice_date` = DATE(`date_created`) WHERE `invoice_date` IS NULL;


-- ============================================================
-- Purchase Date (user-entered date on purchase create / edit)
-- Date: 2026-10-23
-- ============================================================
-- New column on `purchase_list`. Existing purchases get the date
-- they were created. Bill Date (supplier's bill date) is unchanged.
-- Run AFTER the Purchase Master section. Safe to run more than once.
-- ============================================================

ALTER TABLE `purchase_list` ADD COLUMN IF NOT EXISTS `purchase_date` date DEFAULT NULL AFTER `code`;
UPDATE `purchase_list` SET `purchase_date` = DATE(`date_created`) WHERE `purchase_date` IS NULL;

-- ============================================================
-- Stock Report: Opening Stock on products
-- Date: 2026-10-23
-- ============================================================
-- Stock = Opening Stock + Purchases - Invoices (cancelled bills not counted).
-- Only purchases / invoices dated on or after `opening_stock_date`
-- are counted; if it is empty, all history counts.
-- Safe to run more than once.
-- ============================================================

ALTER TABLE `product_list` ADD COLUMN IF NOT EXISTS `opening_stock` int(11) NOT NULL DEFAULT 0 AFTER `price`;
ALTER TABLE `product_list` ADD COLUMN IF NOT EXISTS `opening_stock_date` date DEFAULT NULL AFTER `opening_stock`;
